// Shared-game deterministic one-step faction agenda transition.
// Governed by procedure.game.core.world.faction.
var faction = ctx.roles.faction;
var factionComponent = 'game.core.world.faction';

function closed(value, keys) {
  if (value === null || Array.isArray(value) || typeof value !== 'object') { return false; }
  var actual = Object.keys(value).sort();
  if (actual.length !== keys.length) { return false; }
  for (var index = 0; index < keys.length; index++) {
    if (actual[index] !== keys[index]) { return false; }
  }
  return true;
}

function parse(raw, name) {
  if (typeof raw !== 'string') { throw new Error(name + ' is corrupt.'); }
  try { return JSON.parse(raw); }
  catch (error) { throw new Error(name + ' is corrupt.'); }
}

function text(value, maximum) {
  return typeof value === 'string' && value.length >= 1 && value.trim() === value &&
    Array.from(value).length <= maximum;
}

function textList(value, minimum, maximum) {
  if (!Array.isArray(value) || value.length < minimum || value.length > maximum) { return false; }
  for (var index = 0; index < value.length; index++) {
    if (!text(value[index], 500)) { return false; }
    for (var previous = 0; previous < index; previous++) {
      if (value[previous] === value[index]) { return false; }
    }
  }
  return true;
}

function validFaction(value) {
  return closed(value, ['agenda', 'assets', 'goals', 'methods', 'status', 'summary', 'visibility']) &&
    (value.status === 'draft' || value.status === 'active' || value.status === 'archived') &&
    text(value.summary, 1000) &&
    (value.visibility === 'public' || value.visibility === 'party' || value.visibility === 'gm') &&
    textList(value.goals, 1, 5) && textList(value.methods, 1, 5) && textList(value.assets, 0, 10) &&
    closed(value.agenda, ['state', 'summary']) &&
    (value.agenda.state === 'ready' || value.agenda.state === 'advanced') && text(value.agenda.summary, 1000);
}

if (!closed(ctx.input, [])) {
  throw new Error('Faction agenda input must be exactly {}. Do not supply state, summary, transition, effect, reason, or result.');
}
if (!faction || !faction.components || !faction.components[factionComponent]) {
  throw new Error('Faction agenda advancement requires one faction role carrying game.core.world.faction.');
}

var current = parse(faction.components[factionComponent], 'Faction state');
if (!validFaction(current)) { throw new Error('Faction state is corrupt.'); }
if (current.status !== 'active') { throw new Error('Only an active faction may advance its agenda.'); }
if (current.agenda.state !== 'ready') { throw new Error('Faction agenda is not ready to advance.'); }

var next = {
  status: current.status,
  summary: current.summary,
  visibility: current.visibility,
  goals: current.goals,
  methods: current.methods,
  assets: current.assets,
  agenda: { state: 'advanced', summary: current.agenda.summary }
};

return {
  narration: faction.name + ' advances its agenda.',
  effects: [{ type: 'component.set', entityId: faction.id, definitionId: factionComponent, data: JSON.stringify(next) }],
  data: {
    test: 'faction-agenda-advance',
    factionId: faction.id,
    previousState: 'ready',
    currentState: 'advanced'
  }
};
