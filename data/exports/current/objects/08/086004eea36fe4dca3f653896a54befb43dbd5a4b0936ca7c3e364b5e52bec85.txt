  <script>
    (() => {
      const storageKeys = { notes: "dantes.personal-dashboard.notes.v1", chat: "dantes.personal-dashboard.chat.v1" };
      const dateTime = document.querySelector("#local-date-time");
      const notes = document.querySelector("#personal-notes");
      const notesStatus = document.querySelector("#notes-status");
      const application = document.querySelector("#chat-application");
      const stateSpace = document.querySelector("#chat-state-space");
      const chatState = document.querySelector("#chat-state");
      const chat = document.querySelector("#local-outer-chat");
      const refresh = document.querySelector("#refresh-chat-context");
      let chatPreferences = readStoredJson(storageKeys.chat, {});
      const updateClock = () => {
        const now = new Date(); dateTime.dateTime = now.toISOString();
        dateTime.textContent = new Intl.DateTimeFormat(undefined, { weekday: "long", year: "numeric", month: "long", day: "numeric", hour: "2-digit", minute: "2-digit", second: "2-digit" }).format(now);
      };
      updateClock(); window.setInterval(updateClock, 1000);
      try { notes.value = localStorage.getItem(storageKeys.notes) || ""; }
      catch (_) { setNotesStatus("Browser storage is unavailable. This note will not survive a refresh.", true); }
      notes.addEventListener("input", () => {
        try { localStorage.setItem(storageKeys.notes, notes.value); setNotesStatus("Saved locally in this browser.", false); }
        catch (_) { setNotesStatus("Browser storage is unavailable. This note will not survive a refresh.", true); }
      });
      application.addEventListener("change", async () => { rememberChatPreferences({ applicationId: application.value, stateSpaceId: null }); await loadStateSpaces(); });
      stateSpace.addEventListener("change", () => { rememberChatPreferences({ stateSpaceId: stateSpace.value }); mountConversation(); });
      refresh.addEventListener("click", loadApplications);
      async function loadApplications() {
        setChatState("Loading available applications…"); disableChatSetup(true);
        try {
          const applications = await readAll("/api/control/structure/applications");
          fillSelect(application, applications, "id", item => item.displayName || item.id, chatPreferences.applicationId);
          if (!application.value) { chat.replaceChildren(); setChatState("No registered applications are available yet.", true); return; }
          await loadStateSpaces();
        } catch (_) { chat.replaceChildren(); setChatState("Application structure is unavailable. Open the control center to check local access and registered applications.", true); }
        finally { disableChatSetup(false); }
      }
      async function loadStateSpaces() {
        if (!application.value) return;
        setChatState("Loading state spaces…"); stateSpace.disabled = true;
        try {
          const path = "/api/control/structure/applications/" + encodeURIComponent(application.value) + "/state-spaces";
          const spaces = await readAll(path);
          fillSelect(stateSpace, spaces, "stateSpaceId", item => item.stateSpaceId, chatPreferences.stateSpaceId);
          rememberChatPreferences({ applicationId: application.value, stateSpaceId: stateSpace.value || null });
          if (!stateSpace.value) { chat.replaceChildren(); setChatState("This application has no state spaces yet. Create one from the system workflow, then refresh here.", true); return; }
          mountConversation();
        } catch (_) { chat.replaceChildren(); setChatState("State spaces are unavailable for this application. Refresh or inspect it in the control center.", true); }
        finally { stateSpace.disabled = false; }
      }
      function mountConversation() {
        if (!application.value || !stateSpace.value) return;
        chat.replaceChildren();
        const conversation = document.createElement("application-conversation");
        conversation.setAttribute("application-id", application.value);
        conversation.setAttribute("state-space-id", stateSpace.value);
        conversation.setAttribute("session-context-id", chatPreferences.sessionContextId || createSessionContext());
        conversation.addEventListener("error", event => setChatState(event.detail.message, true));
        conversation.addEventListener("progress", () => setChatState("Local outer chat is working…"));
        conversation.addEventListener("receipt", () => setChatState("Action receipt received. Review the conversation for the outcome."));
        chat.append(conversation); setChatState("Ready for a local outer conversation.");
      }
      async function readAll(path) {
        const values = []; let next = path + (path.includes("?") ? "&" : "?") + "limit=100";
        while (next) {
          const response = await fetch(next, { cache: "no-store", credentials: "same-origin" });
          if (!response.ok) throw new Error("structure-unavailable");
          const page = await response.json(); values.push(...(page.items || []));
          next = page.nextCursor ? path + (path.includes("?") ? "&" : "?") + "limit=100&cursor=" + encodeURIComponent(page.nextCursor) : null;
        }
        return values;
      }
      function fillSelect(select, values, id, label, preferred) {
        select.replaceChildren();
        for (const value of values) { const option = document.createElement("option"); option.value = value[id]; option.textContent = label(value); select.append(option); }
        const match = values.find(value => value[id] === preferred); if (match) select.value = match[id];
      }
      function rememberChatPreferences(change) {
        chatPreferences = { ...chatPreferences, ...change }; if (!chatPreferences.sessionContextId) chatPreferences.sessionContextId = createSessionContext();
        try { localStorage.setItem(storageKeys.chat, JSON.stringify(chatPreferences)); } catch (_) { }
      }
      function createSessionContext() {
        const random = crypto.randomUUID ? crypto.randomUUID().replaceAll("-", "") : Date.now().toString(36);
        const sessionContextId = "home-session." + random; chatPreferences.sessionContextId = sessionContextId; return sessionContextId;
      }
      function readStoredJson(key, fallback) { try { return JSON.parse(localStorage.getItem(key) || "null") || fallback; } catch (_) { return fallback; } }
      function disableChatSetup(disabled) { application.disabled = disabled; stateSpace.disabled = disabled; refresh.disabled = disabled; }
      function setChatState(message, isError = false) { chatState.textContent = message; chatState.classList.toggle("error", isError); }
      function setNotesStatus(message, isError) { notesStatus.textContent = message; notesStatus.classList.toggle("error", isError); }
      loadApplications();
    })();
  </script>
