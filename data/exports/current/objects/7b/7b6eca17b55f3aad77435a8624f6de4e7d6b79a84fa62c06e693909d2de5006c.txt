var rumour = ctx.roles.rumour, world = ctx.roles.world, rumourComponent = 'game.core.world.rumour', scopeKind = 'game.core.world.knowledge.in-world';
function closed(v, keys) { if (v === null || Array.isArray(v) || typeof v !== 'object') return false; var a = Object.keys(v).sort(); if (a.length !== keys.length) return false; for (var i=0;i<keys.length;i++) if (a[i] !== keys[i]) return false; return true; }
function parse(v) { if (typeof v !== 'string') throw new Error('Rumour state is corrupt.'); try { return JSON.parse(v); } catch (e) { throw new Error('Rumour state is corrupt.'); } }
function text(v,n) { return typeof v === 'string' && v.length >= 1 && v.trim() === v && Array.from(v).length <= n; }
if (!closed(ctx.input, [])) throw new Error('Rumour confirmation input must be exactly {}.');
if (!rumour || !world || !rumour.components || !rumour.components[rumourComponent]) throw new Error('Rumour confirmation requires rumour and world roles.');
var current = parse(rumour.components[rumourComponent]);
if (!closed(current,['provenance','status','summary','visibility']) || !text(current.summary,1000) || !text(current.provenance,500) || ['unconfirmed','confirmed','disproved','archived'].indexOf(current.status) < 0 || ['public','party','gm'].indexOf(current.visibility) < 0) throw new Error('Rumour state is corrupt.');
if (current.status !== 'unconfirmed') throw new Error('Only an unconfirmed rumour may be confirmed.');
if (!Array.isArray(rumour.relationships)) throw new Error('Rumour scope projection is missing.');
var count=0; for (var i=0;i<rumour.relationships.length;i++) { var e=rumour.relationships[i]; if (!closed(e,['data','fromEntityId','kind','toEntityId']) || typeof e.data !== 'string') throw new Error('Rumour relationship projection is corrupt.'); if (e.kind === scopeKind) { if (e.fromEntityId !== rumour.id || e.toEntityId !== world.id || !closed(parse(e.data),[])) throw new Error('Rumour has corrupt scope state.'); count++; } }
if (count !== 1) throw new Error('Rumour must have exactly one stored scope link to the supplied world.');
var next={status:'confirmed',summary:current.summary,provenance:current.provenance,visibility:current.visibility};
return { narration: rumour.name + ' is confirmed.', effects:[{type:'component.set',entityId:rumour.id,definitionId:rumourComponent,data:JSON.stringify(next)}], data:{test:'world-rumour-confirm',rumourId:rumour.id,previousStatus:'unconfirmed',currentStatus:'confirmed'} };
