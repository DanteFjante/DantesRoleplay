  <script>
    (() => {
      "use strict";

      const panels = {
        "server-settings": { title: "Server settings", waiting: "Loading settings availability…" },
        "effect-history": { title: "Past effects", waiting: "Loading effect-history availability…" },
        "assistant": { title: "Assistants", waiting: "Loading assistant availability…" },
        "ecs-explorer": { title: "ECS & contracts", waiting: "Loading ECS availability…" },
        "site-editor": { title: "Site editor", waiting: "Loading site-editor availability…" }
      };
      const workspaceRoutes = Object.freeze({
        settings: { panelId: "server-settings", title: "Server settings", navRoute: "settings" },
        effects: { panelId: "effect-history", title: "Past effects", navRoute: "effects" },
        assistants: { panelId: "assistant", title: "Assistants", navRoute: "assistants" },
        applications: { panelId: "ecs-explorer", title: "Applications", navRoute: "applications" },
        "site-editor": { panelId: "site-editor", title: "Site editor", navRoute: "site-editor" }
      });
      let statusRequest;

      function parseWorkspaceRoute(hash) {
        if (typeof hash !== "string" || !hash.startsWith("#/") || hash.length > 640) return null;
        const routeText = hash.slice(2);
        if (workspaceRoutes[routeText]) return { ...workspaceRoutes[routeText], applicationId: null };
        if (!routeText.startsWith("applications/")) return null;
        const encodedId = routeText.slice("applications/".length);
        if (!encodedId || encodedId.includes("/")) return null;
        try {
          const applicationId = decodeURIComponent(encodedId);
          if (!applicationId || applicationId.length > 200 || applicationId.includes("/") ||
              applicationId.includes("\\") || Array.from(applicationId).some(character =>
                character.charCodeAt(0) < 32 || character.charCodeAt(0) === 127)) return null;
          return {
            ...workspaceRoutes.applications,
            title: "Application · " + applicationId,
            applicationId
          };
        } catch (_) { return null; }
      }

      function navigateToApplication(applicationId) {
        const target = "#/applications/" + encodeURIComponent(applicationId);
        if (window.location.hash === target) renderRoute();
        else window.location.hash = target;
      }

      function renderRoute() {
        let route = parseWorkspaceRoute(window.location.hash);
        if (!route) {
          route = { ...workspaceRoutes.settings, applicationId: null };
          if (window.location.hash !== "#/settings") history.replaceState(null, "", "#/settings");
        }
        document.querySelectorAll("[data-workspace]").forEach(workspace => {
          workspace.hidden = workspace.id !== route.panelId;
        });
        document.querySelectorAll("[data-route]").forEach(link => {
          if (link.dataset.route === route.navRoute) link.setAttribute("aria-current", "page");
          else link.removeAttribute("aria-current");
        });
        document.getElementById("workspace-title").textContent = route.title;
        document.title = route.title + " — Dante's Roleplay Control Center";
        const explorer = document.getElementById("ecs-explorer");
        if (route.panelId === "ecs-explorer" && route.applicationId)
          explorer.openApplicationFromRoute(route.applicationId);
        else if (route.panelId === "ecs-explorer")
          explorer.showApplicationListFromRoute();
      }

      function loadStatus(retry) {
        if (retry || !statusRequest) {
          statusRequest = fetch("/api/control/status", { cache: "no-store", credentials: "same-origin" })
            .then(async response => {
              if (!response.ok) throw new Error(response.status === 403 ? "forbidden" : "unavailable");
              const document = await response.json();
              if (!document || document.status !== "ready" || !Array.isArray(document.panels)) {
                throw new Error("malformed");
              }
              return document;
            });
        }
        return statusRequest;
      }

      function setOverview(status, failure) {
        const heading = window.document.getElementById("overview-heading");
        const detail = window.document.getElementById("overview-detail");
        const state = window.document.getElementById("overview-state");
        if (failure) {
          heading.textContent = failure === "forbidden" ? "Control access is not available" : "Control status is unavailable";
          detail.textContent = "Panel navigation remains available. Retry an individual panel when access is restored.";
          state.textContent = failure;
          state.classList.add("error");
          return;
        }
        heading.textContent = "Control boundary ready";
        detail.textContent = status.access.mode === "tailscale"
          ? "Connected through the approved private Tailscale web route."
          : "Connected from this computer through the local web route.";
        state.textContent = status.access.mode;
      }

      class ControlPanel extends HTMLElement {
        connectedCallback() {
          if (this.dataset.initialized) return;
          this.dataset.initialized = "true";
          this.kind = this.id;
          const copy = panels[this.kind];
          this.innerHTML = '<section class="panel" aria-labelledby="' + this.kind + '-heading">' +
            '<div class="panel-head"><div><p class="eyebrow">Control panel</p><h2 id="' + this.kind + '-heading"></h2></div>' +
            '<span class="status" data-state>loading</span></div><p class="panel-body" data-message></p>' +
            '<button type="button" data-retry hidden>Retry</button></section>';
          this.querySelector("h2").textContent = copy.title;
          this.querySelector("[data-retry]").addEventListener("click", () => this.refresh(true));
          this.refresh(false);
        }

        async refresh(retry) {
          const copy = panels[this.kind];
          this.render("loading", copy.waiting, false);
          try {
            const document = await loadStatus(retry);
            const panel = document.panels.find(candidate => candidate && candidate.id === this.kind);
            if (!panel || typeof panel.state !== "string" || typeof panel.message !== "string") {
              throw new Error("malformed");
            }
            this.render(panel.state, panel.message, false);
            setOverview(document, null);
          } catch (error) {
            const state = error && error.message === "forbidden" ? "forbidden" : "unavailable";
            const message = state === "forbidden"
              ? "This browser is not authorized to read control status."
              : "Control status could not be loaded. This panel can be retried without affecting the others.";
            this.render(state, message, true);
            setOverview(document, state);
          }
        }

        render(state, message, showRetry) {
          this.querySelector("[data-state]").textContent = state;
          this.querySelector("[data-message]").textContent = message;
          this.querySelector("[data-retry]").hidden = !showRetry;
        }
      }

      class ServerSettingsPanel extends ControlPanel {
        async refresh(retry) {
          this.render("loading", "Loading host setting definitions…", false);
          try {
            const status = await loadStatus(retry);
            const response = await fetch("/api/control/settings", {
              cache: "no-store", credentials: "same-origin"
            });
            if (!response.ok) throw new Error(response.status === 403 ? "forbidden" : "unavailable");
            const page = await response.json();
            if (!page || !Array.isArray(page.items)) throw new Error("malformed");
            this.renderSettings(page);
            setOverview(status, null);
          } catch (error) {
            const state = error && error.message === "forbidden" ? "forbidden" : "unavailable";
            this.render(state, state === "forbidden"
              ? "This browser is not authorized to read server settings."
              : "Server settings could not be loaded. This panel can be retried without affecting the others.", true);
          }
        }

        renderSettings(page) {
          const body = this.querySelector("[data-message]");
          body.replaceChildren();
          this.querySelector("[data-state]").textContent = page.state || "ready";
          this.querySelector("[data-retry]").hidden = true;
          body.append(this.text(page.message || "Host setting definitions are available."));
          body.append(this.text("Changes are staged safely and take effect the next time the server starts."));
          const list = this.section(body, "Local completion startup settings");
          (page.items || []).forEach(item => list.append(this.choice(
            item.displayName,
            this.valueLabel(item) + " · " + item.source + " · " + item.mutability,
            () => this.inspect(item.key, body))));
          if (!page.items || page.items.length === 0) list.append(this.text("No host settings are registered."));
        }

        async inspect(key, body) {
          body.querySelectorAll("[data-setting-detail]").forEach(node => node.closest("section")?.remove());
          const detail = this.section(body, "Setting detail");
          detail.dataset.settingDetail = "true";
          detail.append(this.text("Loading exact definition…"));
          try {
            const response = await fetch("/api/control/settings/" + encodeURIComponent(key), {
              cache: "no-store", credentials: "same-origin"
            });
            if (!response.ok) throw new Error("unavailable");
            const exact = await response.json();
            const item = exact.summary;
            detail.replaceChildren();
            const label = document.createElement("label");
            label.className = "setting-field";
            const caption = document.createElement("span");
            caption.textContent = item.displayName;
            const input = document.createElement("input");
            input.disabled = item.sensitivity !== "public-value" || item.mutability === "read-only";
            input.value = this.editValue(item.pendingValue ?? item.value);
            label.append(caption, input);
            const actions = document.createElement("div");
            actions.className = "toolbar";
            const save = document.createElement("button");
            save.type = "button";
            save.textContent = "Stage change";
            save.disabled = input.disabled;
            save.addEventListener("click", () => this.mutate(key, "", "PUT", {
              expectedRevision: item.revision, value: this.parseValue(input.value, exact.schema)
            }, detail));
            const reset = document.createElement("button");
            reset.type = "button";
            reset.textContent = "Reset to startup value";
            reset.disabled = input.disabled || item.revision === 0;
            reset.addEventListener("click", () => this.mutate(key, "/reset", "POST", {
              expectedRevision: item.revision
            }, detail));
            const history = document.createElement("button");
            history.type = "button";
            history.textContent = "Show history";
            history.addEventListener("click", () => this.history(key, item, detail));
            actions.append(save, reset, history);
            detail.append(
              label,
              actions,
              this.text(item.description),
              this.text("Sensitivity: " + item.sensitivity + " · Mutability: " + item.mutability +
                " · Disruption: " + item.disruption + " · Revision: " + item.revision +
                " · Applied: " + item.appliedRevision + " · Restart pending: " + (item.restartRequired ? "yes" : "no")),
              this.pre("JSON Schema\n" + JSON.stringify(exact.schema, null, 2)));
          } catch (_) {
            detail.replaceChildren(this.text("The exact setting definition is unavailable. Select it again to retry."));
          }
        }

        valueLabel(item) {
          if (item.sensitivity === "configured-only") return item.configured ? "Configured" : "Not configured";
          if (item.value === null || typeof item.value === "undefined") return "No exposed value";
          return typeof item.value === "string" ? item.value : JSON.stringify(item.value);
        }

        editValue(value) {
          if (value === null || typeof value === "undefined") return "";
          return typeof value === "string" ? value : JSON.stringify(value);
        }

        parseValue(raw, schema) {
          if (schema.type === "boolean") {
            if (raw === "true") return true;
            if (raw === "false") return false;
            throw new Error("Enter true or false.");
          }
          if (schema.type === "integer") {
            const value = Number(raw);
            if (!Number.isInteger(value)) throw new Error("Enter a whole number.");
            return value;
          }
          return raw;
        }

        async mutate(key, suffix, method, body, detail) {
          try {
            const response = await fetch("/api/control/settings/" + encodeURIComponent(key) + suffix, {
              method, credentials: "same-origin", headers: { "Content-Type": "application/json" },
              body: JSON.stringify(body)
            });
            const result = await response.json();
            if (!response.ok) throw new Error(result.message || "The setting change was rejected.");
            detail.append(this.text("Revision " + result.revision + " staged. Restart the server when convenient."));
            await this.refresh(true);
          } catch (error) {
            detail.append(this.text(error.message || "The setting change failed."));
          }
        }

        async history(key, item, detail) {
          try {
            const response = await fetch("/api/control/settings/" + encodeURIComponent(key) + "/versions", {
              cache: "no-store", credentials: "same-origin"
            });
            if (!response.ok) throw new Error("Setting history is unavailable.");
            const page = await response.json();
            const section = this.section(detail, "Revision history");
            (page.items || []).forEach(version => {
              const row = document.createElement("div");
              row.className = "explorer-choice";
              const summary = document.createElement("span");
              summary.textContent = "Revision " + version.version + " · " + version.state + " · " +
                (version.isReset ? "startup value" : this.editValue(version.value)) + " · " + version.createdBy;
              row.append(summary);
              if (version.version < item.revision) {
                const rollback = document.createElement("button");
                rollback.type = "button";
                rollback.textContent = "Roll back to this revision";
                rollback.addEventListener("click", event => {
                  event.stopPropagation();
                  if (confirm("Stage revision " + version.version + " as the next setting value?"))
                    this.mutate(key, "/rollback", "POST", {
                      expectedRevision: item.revision, targetRevision: version.version
                    }, detail);
                });
                row.append(rollback);
              }
              section.append(row);
            });
            if (!page.items || page.items.length === 0) section.append(this.text("No override history yet."));
          } catch (error) { detail.append(this.text(error.message || "Setting history is unavailable.")); }
        }

        section(parent, title) {
          const wrapper = document.createElement("section");
          wrapper.className = "history-group";
          const heading = document.createElement("h3");
          heading.textContent = title;
          wrapper.append(heading);
          const content = document.createElement("div");
          content.className = "explorer-list";
          wrapper.append(content);
          parent.append(wrapper);
          return content;
        }

        choice(label, metaText, action) {
          const button = document.createElement("button");
          button.type = "button";
          button.className = "explorer-choice";
          button.textContent = label;
          const meta = document.createElement("small");
          meta.textContent = metaText;
          button.append(meta);
          button.addEventListener("click", action);
          return button;
        }

        text(value) { const node = document.createElement("p"); node.textContent = value; return node; }
        pre(value) { const node = document.createElement("pre"); node.textContent = value; return node; }
      }

      class EffectHistoryPanel extends ControlPanel {
        connectedCallback() {
          super.connectedCallback();
          this.cursor = null;
        }

        async refresh(retry) {
          if (retry) this.cursor = null;
          this.render("loading", "Loading committed effects…", false);
          try {
            const status = await loadStatus(retry);
            const response = await fetch("/api/control/effects" + (this.cursor ? "?cursor=" + encodeURIComponent(this.cursor) : ""), {
              cache: "no-store", credentials: "same-origin"
            });
            if (!response.ok) throw new Error(response.status === 403 ? "forbidden" : "unavailable");
            const page = await response.json();
            if (!page || !Array.isArray(page.groups)) throw new Error("malformed");
            this.renderHistory(page);
            setOverview(status, null);
          } catch (error) {
            const state = error && error.message === "forbidden" ? "forbidden" : "unavailable";
            this.render(state, state === "forbidden"
              ? "This browser is not authorized to read past effects."
              : "Effect history could not be loaded. This panel can be retried without affecting the others.", true);
          }
        }

        renderHistory(page) {
          const body = this.querySelector("[data-message]");
          const retry = this.querySelector("[data-retry]");
          body.replaceChildren();
          this.querySelector("[data-state]").textContent = "ready";
          retry.hidden = true;
          if (page.groups.length === 0) {
            body.textContent = "No accepted effects have been recorded yet.";
            return;
          }
          page.groups.forEach(group => {
            const section = document.createElement("section");
            section.className = "history-group";
            const label = document.createElement("p");
            label.textContent = "Operation " + group.rootOperationId;
            section.append(label);

        async write(path, body) {
          const response = await fetch(path, {
            method: "POST", credentials: "same-origin", headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body)
          });
          const document = await response.json();
          if (!response.ok) throw new Error(document.message || "The assistant request was rejected.");
          return document;
        }

        async writeStream(path, body, onEvent) {
          const response = await fetch(path, {
            method: "POST", credentials: "same-origin", headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body)
          });
          if (!response.ok) {
            const failure = await response.json().catch(() => ({}));
            throw new Error(failure.message || "The Codex request was rejected.");
          }
          if (!response.body) throw new Error("The Codex stream is unavailable.");
          const reader = response.body.getReader();
          const decoder = new TextDecoder();
          let buffer = "";
          let finalDocument = null;
          while (true) {
            const part = await reader.read();
            buffer += decoder.decode(part.value || new Uint8Array(), { stream: !part.done });
            if (buffer.length > 300000) throw new Error("A Codex stream event exceeded its bound.");
            let newline;
            while ((newline = buffer.indexOf("\n")) >= 0) {
              const line = buffer.slice(0, newline).trim();
              buffer = buffer.slice(newline + 1);
              if (!line) continue;
              const event = JSON.parse(line);
              onEvent(event);
              if (event.conversation) finalDocument = event.conversation;
            }
            if (part.done) break;
          }
          if (buffer.trim()) {
            const event = JSON.parse(buffer);
            onEvent(event);
            if (event.conversation) finalDocument = event.conversation;
          }
          if (!finalDocument) throw new Error("Codex ended without a conversation result.");
          return finalDocument;
        }

        newKey() {
          return "web:" + (crypto.randomUUID ? crypto.randomUUID() : Date.now() + ":" + Math.random());
        }
        section(parent, title) { const section = document.createElement("section"); section.className = "history-group"; const heading = document.createElement("h3"); heading.textContent = title; section.append(heading); parent.append(section); return section; }
        choice(label, meta, action) { const button = document.createElement("button"); button.type = "button"; button.className = "explorer-choice"; button.textContent = label; const small = document.createElement("small"); small.textContent = meta; button.append(small); button.addEventListener("click", action); return button; }
        text(value) { const node = document.createElement("p"); node.textContent = value; return node; }
      }

              "Value (revision " + component.revision + ")\n" + prettyJson(component.valueJson) +
              "\n\nSchema (version " + schema.version + ", hash " + schema.schemaHash + ")\n" + prettyJson(schema.schemaJson)));
          } catch (_) { detail.replaceChildren(this.empty("The component evidence is unavailable. Select it again to retry.")); }
        }

        async browseCatalog(applicationId, collection, branch, body) {
          const detail = this.catalogDetail(body, "Published contracts");
          try {
            const path = "/api/control/structure/applications/" + encodeURIComponent(applicationId) +
              "/catalog/browse?collection=" + encodeURIComponent(collection) +
              "&branch=" + encodeURIComponent(branch);
            const page = await this.read(path);
            detail.replaceChildren();
            (page.entries || []).forEach(entry => {
              if (entry.node) {
                detail.append(this.choice(
                  entry.node.title || entry.node.path,
                  "Collection branch",
                  () => this.browseCatalog(applicationId, collection, entry.node.path, body)));
              } else if (entry.record) {
                detail.append(this.choice(
                  entry.record.name || entry.record.qualifiedId,
                  entry.record.kind + " · " + entry.record.qualifiedId,
                  () => this.inspectCatalog(applicationId, collection, entry.record.qualifiedId, body)));
              }
            });
            if (!page.entries || page.entries.length === 0) detail.append(this.empty("This public branch is empty."));
          } catch (_) { detail.replaceChildren(this.empty("Published contracts are unavailable. Choose the collection again to retry.")); }
        }

        async searchCatalog(applicationId, query, body) {
          const detail = this.catalogDetail(body, "Public catalog search");
          try {
            const path = "/api/control/structure/applications/" + encodeURIComponent(applicationId) +
              "/catalog/search?q=" + encodeURIComponent(query);
            const page = await this.read(path);
            detail.replaceChildren();
            (page.records || []).forEach(hit => detail.append(this.choice(
              hit.record.name || hit.record.qualifiedId,
              hit.record.kind + " · " + hit.record.qualifiedId,
              () => this.inspectCatalog(applicationId, hit.record.collection, hit.record.qualifiedId, body))));
            if (!page.records || page.records.length === 0) detail.append(this.empty("No public contracts matched."));
          } catch (_) { detail.replaceChildren(this.empty("Catalog search is unavailable. Try the search again.")); }
        }

        async inspectCatalog(applicationId, collection, qualifiedId, body) {
          const detail = this.catalogDetail(body, "Published contract detail");
          try {
            const path = "/api/control/structure/applications/" + encodeURIComponent(applicationId) +
              "/catalog/records/" + encodeURIComponent(qualifiedId) + "?collection=" + encodeURIComponent(collection);
            const record = await this.read(path);
            detail.replaceChildren(this.pre(
              record.summary.name + " · version " + record.summary.version +
              "\n" + record.summary.description + "\n\n" + prettyJson(record.contentJson)));
          } catch (_) { detail.replaceChildren(this.empty("The published contract is unavailable. Select it again to retry.")); }
        }

        section(parent, title, nested) {
          const wrapper = document.createElement("section");
          wrapper.className = nested ? "explorer-section" : "history-group";
          const heading = document.createElement("h3");
          heading.textContent = title;
          wrapper.append(heading);
          const list = document.createElement("div");
          list.className = "explorer-list";
          wrapper.append(list);
          parent.append(wrapper);
          return list;
        }

        choice(label, metaText, action) {
          const button = document.createElement("button");
          button.type = "button";
          button.className = "explorer-choice";
          button.textContent = label;
          const meta = document.createElement("small");
          meta.textContent = metaText;
          button.append(meta);
          button.addEventListener("click", action);
          return button;
        }

        empty(message) { const node = document.createElement("p"); node.textContent = message; return node; }
        pre(value) { const node = document.createElement("pre"); node.textContent = value; return node; }
        clearDetails(body) { body.querySelectorAll("[data-detail], .explorer-grid, .history-group:not(:first-of-type)").forEach(node => node.remove()); }
        removeAfter(body, marker) {
          const targets = marker === "entities"
            ? ["entities", "components", "component-detail"]
            : ["components", "component-detail"];
          body.querySelectorAll("[data-detail]").forEach(node => {
            if (targets.includes(node.dataset.detail)) node.closest("section")?.remove();
          });
        }
        detail(body, title) {
          body.querySelectorAll("[data-detail='component-detail']").forEach(node => node.closest("section")?.remove());
          const list = this.section(body, title);
          list.dataset.detail = "component-detail";
          list.append(this.empty("Loading exact evidence…"));
          return list;
        }
        catalogDetail(body, title) {
          body.querySelectorAll("[data-detail='catalog-detail']").forEach(node => node.closest("section")?.remove());
          const list = this.section(body, title);
          list.dataset.detail = "catalog-detail";
          list.append(this.empty("Loading public catalog…"));
          return list;
        }
      }

      class SiteEditorPanel extends ControlPanel {
        async refresh(retry) {
          this.render("loading", "Loading editable pages…", false);
          try {
            const status = await loadStatus(retry);
            const page = await this.read("/api/control/pages");
            this.renderPages(page);
            setOverview(status, null);
          } catch (error) {
            const state = error && error.message === "forbidden" ? "forbidden" : "unavailable";
            this.render(state, state === "forbidden"
              ? "This browser is not authorized to edit pages."
              : "The page editor could not be loaded. This panel can be retried without affecting the others.", true);
          }
        }

        async read(path) {
          const response = await fetch(path, { cache: "no-store", credentials: "same-origin" });
          if (!response.ok) throw new Error(response.status === 403 ? "forbidden" : "unavailable");
          return response.json();
        }

        async write(path, method, body) {
          const response = await fetch(path, {
            method,
            cache: "no-store",
            credentials: "same-origin",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body)
          });
          if (!response.ok) {
            let error;
            try { error = await response.json(); } catch (_) { error = null; }
            const failure = new Error(error && error.message ? error.message : "The page change was rejected.");
            failure.code = error && error.error ? error.error : "unavailable";
            throw failure;
          }
          return response.json();
        }

        renderPages(page) {
          const body = this.querySelector("[data-message]");
          body.replaceChildren();
          this.querySelector("[data-state]").textContent = "ready";
          this.querySelector("[data-retry]").hidden = true;
          const intro = document.createElement("p");
          intro.textContent = "Choose an existing page. Saving creates an inactive revision; preview and publishing are separate actions.";
          body.append(intro);
          const list = this.section(body, "Pages");
          if (!page.items || page.items.length === 0) {
            list.append(this.text("No pages exist yet. Create the first revision through the direct upload route."));
            return;
          }
          page.items.forEach(item => {
            list.append(this.choice(
              item.id,
              "active " + item.activeRevision + " · latest " + item.latestRevision,
              () => this.selectPage(item.id, body)));
            list.append(this.livePageLink(item.id));
          });
        }

        async selectPage(pageId, body) {
          this.clearPage(body);
          const loading = this.section(body, "Page detail");
          loading.dataset.siteDetail = "page";
          loading.append(this.text("Loading immutable revisions…"));
          try {
            const root = "/api/control/pages/" + encodeURIComponent(pageId);
            const [page, revisions] = await Promise.all([this.read(root), this.read(root + "/revisions")]);
            this.clearPage(body);
            const header = this.section(body, page.id);
            header.dataset.siteDetail = "page";
            header.append(this.text("Active revision " + page.activeRevision + "; latest revision " + page.latestRevision + "."));
            if (page.id === "control-center") {
              const recovery = this.text("Recovery: keep a local copy. If this editor becomes unusable, upload it directly through /api/pages/control-center or /api/pages/control-center/bundle.");
              recovery.className = "recovery-note";
              header.append(recovery);
            }
            const list = this.section(body, "Immutable revisions");
            list.dataset.siteDetail = "revisions";
            (revisions.items || []).forEach(revision => list.append(this.choice(
              "Revision " + revision.revision + (revision.isActive ? " · active" : " · draft/history"),
              new Date(revision.createdAtUtc).toLocaleString() + " · " + revision.assetCount + " asset(s)",
              () => this.selectRevision(page, revision.revision, body))));
            if (!revisions.items || revisions.items.length === 0) list.append(this.text("No revisions are available."));
          } catch (_) {
            loading.replaceChildren(this.text("This page is unavailable. Choose it again to retry."));
          }
        }

        async selectRevision(page, revision, body) {
          this.clearEditor(body);
          const editor = this.section(body, "Edit revision " + revision);
          editor.dataset.siteDetail = "editor";
          editor.append(this.text("Loading exact HTML…"));
          try {
            const root = "/api/control/pages/" + encodeURIComponent(page.id) + "/revisions/" + revision;
            const detail = await this.read(root);
            editor.replaceChildren();
            const metadata = this.text(
              "HTML hash " + detail.summary.contentHash + " · " + detail.assets.length + " asset(s). Saving preserves this exact asset set.");
            const textarea = document.createElement("textarea");
            textarea.className = "editor-area";
            textarea.setAttribute("aria-label", "Page HTML");
            textarea.spellcheck = false;
            textarea.value = detail.html;
            const state = this.text("");
            state.setAttribute("aria-live", "polite");
            const actions = document.createElement("div");
            actions.className = "editor-actions";
            const save = this.button("Save inactive draft", () => this.saveDraft(page, revision, textarea.value, body, state));
            const previewChanges = this.button("Save & preview draft", () =>
              this.saveDraft(page, revision, textarea.value, body, state, true));
            const preview = this.button("Preview saved revision", () => this.preview(page.id, revision, body));
            actions.append(save, previewChanges, preview);
            if (revision !== page.activeRevision) {
              actions.append(this.button(
                revision < page.activeRevision ? "Roll back to this revision" : "Publish this revision",
                () => this.activate(page, revision, body, state)));
            }
            const download = document.createElement("a");
            download.href = root + "/bundle";
            download.download = page.id + "-revision-" + revision + ".zip";
            download.textContent = "Download ZIP";
            actions.append(download);
            editor.append(metadata, textarea, actions, state);
          } catch (_) {
            editor.replaceChildren(this.text("The selected revision is unavailable. Select it again to retry."));
          }
        }

        async saveDraft(page, baseRevision, html, body, state, previewAfterSave = false) {
          state.className = "";
          state.textContent = "Saving an inactive revision…";
          try {
            const detail = await this.write(
              "/api/control/pages/" + encodeURIComponent(page.id) + "/drafts",
              "POST",
              { expectedLatestRevision: page.latestRevision, baseRevision, html });
            if (previewAfterSave) {
              const current = await this.read("/api/control/pages/" + encodeURIComponent(page.id));
              await this.selectRevision(current, detail.summary.revision, body);
              this.preview(page.id, detail.summary.revision, body);
              return;
            }
            state.textContent = "Draft revision " + detail.summary.revision + " saved. The active page did not change.";
            await this.selectPage(page.id, body);
          } catch (error) {
            state.className = "error";
            state.textContent = error && error.code && error.code.includes("STALE")
              ? "The page changed elsewhere. Reload the page and apply your edit to the latest revision."
              : (error.message || "The draft could not be saved.");
          }
        }

        async activate(page, revision, body, state) {
          const action = revision < page.activeRevision ? "roll back" : "publish";
          if (!window.confirm("Confirm " + action + " of " + page.id + " revision " + revision + "?")) return;
          state.className = "";
          state.textContent = "Updating the active revision…";
          try {
            await this.write(
              "/api/control/pages/" + encodeURIComponent(page.id) + "/active",
              "PUT",
              { expectedActiveRevision: page.activeRevision, revision });
            await this.selectPage(page.id, body);
          } catch (error) {
            state.className = "error";
            state.textContent = error && error.code && error.code.includes("STALE")
              ? "The active page changed elsewhere. Reload before trying again."
              : (error.message || "The active revision could not be changed.");
          }
        }

        preview(pageId, revision, body) {
          body.querySelectorAll("[data-site-detail='preview']").forEach(node => node.closest("section")?.remove());
          const preview = this.section(body, "Isolated preview");
          preview.dataset.siteDetail = "preview";
          const note = this.text("Preview scripts run without a same-origin identity and cannot connect to control APIs or external services.");
          const frame = document.createElement("iframe");
          frame.className = "preview-frame";
          frame.title = "Isolated page preview";
          frame.setAttribute("sandbox", "allow-scripts");
          frame.src = "/api/control/pages/" + encodeURIComponent(pageId) + "/revisions/" + revision + "/preview/index.html";
          preview.append(note, frame);
        }

        section(parent, title) {
          const wrapper = document.createElement("section");
          wrapper.className = "history-group";
          const heading = document.createElement("h3");
          heading.textContent = title;
          wrapper.append(heading);
          const content = document.createElement("div");
          content.className = "explorer-list";
          wrapper.append(content);
          parent.append(wrapper);
          return content;
        }

        choice(label, metaText, action) {
          const button = this.button(label, action);
          button.className = "explorer-choice";
          const meta = document.createElement("small");
          meta.textContent = metaText;
          button.append(meta);
          return button;
        }

        button(label, action) {
          const button = document.createElement("button");
          button.type = "button";
          button.textContent = label;
          button.addEventListener("click", action);
          return button;
        }

        livePageLink(pageId) {
          const link = document.createElement("a");
          link.className = "live-page-link";
          link.href = "/ui/" + encodeURIComponent(pageId) + "/index.html";
          link.target = "_blank";
          link.rel = "noopener";
          link.textContent = "Open live page";
          link.setAttribute("aria-label", "Open live page " + pageId);
          return link;
        }

        text(value) { const node = document.createElement("p"); node.textContent = value; return node; }
        clearPage(body) { body.querySelectorAll("[data-site-detail]").forEach(node => node.closest("section")?.remove()); }
        clearEditor(body) { body.querySelectorAll("[data-site-detail='editor'], [data-site-detail='preview']").forEach(node => node.closest("section")?.remove()); }
      }

      function prettyJson(value) {
        try { return JSON.stringify(JSON.parse(value), null, 2); }
        catch (_) { return value || ""; }
      }

      [
        "server-settings-panel",
        "effect-history-panel",
        "assistant-panel",
        "ecs-explorer",
        "site-editor"
      ].forEach(tag => {
        customElements.define(tag,
          tag === "server-settings-panel" ? ServerSettingsPanel :
          tag === "effect-history-panel" ? EffectHistoryPanel :
          tag === "assistant-panel" ? AssistantPanel :
          tag === "ecs-explorer" ? EcsExplorerPanel :
          tag === "site-editor" ? SiteEditorPanel : class extends ControlPanel {});
      });
      window.addEventListener("hashchange", renderRoute);
      renderRoute();
    })();
  </script>
