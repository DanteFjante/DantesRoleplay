// Administrative Hit Point recording for D&D 2024 creatures.
// Governed by procedure.mechanic.dnd2024.hit-points.
var subject = ctx.roles.subject;
var keys = Object.keys(ctx.input).sort();

if (keys.length !== 3 || keys[0] !== 'current' || keys[1] !== 'maximum' || keys[2] !== 'mode') {
  throw new Error('Input must contain exactly {"mode":"record"|"correct","current":<safe integer>,"maximum":<positive safe integer>}. Do not supply sourceRef, damage, healing, deltas, or effects.');
}

var mode = ctx.input.mode;
if (mode !== 'record' && mode !== 'correct') {
  throw new Error('input.mode must be exactly "record" or "correct".');
}

var current = ctx.input.current;
var maximum = ctx.input.maximum;
var maxSafe = 9007199254740991;
if (typeof current !== 'number' || !isFinite(current) || Math.floor(current) !== current || current < 0 || current > maxSafe) {
  throw new Error('input.current must be a safe integer from 0 through 9007199254740991. It is never rounded, parsed, or treated as a delta.');
}
if (typeof maximum !== 'number' || !isFinite(maximum) || Math.floor(maximum) !== maximum || maximum < 1 || maximum > maxSafe) {
  throw new Error('input.maximum must be a positive safe integer from 1 through 9007199254740991. It is never rounded or parsed.');
}
if (current > maximum) {
  throw new Error('input.current cannot exceed input.maximum. Record the complete bounded pair, not damage or healing.');
}

var sourceRef = {
  sourceId: 'source.dnd2024.srd-5.2.1',
  locator: 'Playing the Game > Damage and Healing > Hit Points'
};
var raw = subject.components['dnd2024.hit-points'];
var previousPair = null;

if (raw) {
  var previous;
  try {
    previous = JSON.parse(raw);
  } catch (error) {
    throw new Error('The existing Hit Point component is corrupt and cannot be corrected by this rule. Use a governed migration.');
  }

  if (previous === null || Array.isArray(previous) || Object.keys(previous).length !== 3 ||
      !Object.prototype.hasOwnProperty.call(previous, 'current') ||
      !Object.prototype.hasOwnProperty.call(previous, 'maximum') ||
      !Object.prototype.hasOwnProperty.call(previous, 'sourceRef') ||
      typeof previous.current !== 'number' || !isFinite(previous.current) ||
      Math.floor(previous.current) !== previous.current || previous.current < 0 || previous.current > maxSafe ||
      typeof previous.maximum !== 'number' || !isFinite(previous.maximum) ||
      Math.floor(previous.maximum) !== previous.maximum || previous.maximum < 1 || previous.maximum > maxSafe ||
      previous.current > previous.maximum ||
      previous.sourceRef === null || Array.isArray(previous.sourceRef) ||
      previous.sourceRef.sourceId !== sourceRef.sourceId || previous.sourceRef.locator !== sourceRef.locator ||
      Object.keys(previous.sourceRef).length !== 2) {
    throw new Error('The existing Hit Point component has an invalid shape and cannot be corrected by this rule. Use a governed migration.');
  }

  previousPair = { current: previous.current, maximum: previous.maximum };
}

if (mode === 'record' && raw) {
  throw new Error('Hit Points are already recorded. Use mode "correct" to replace the authoritative bounded pair.');
}
if (mode === 'correct' && !raw) {
  throw new Error('Hit Points are absent. Use mode "record" to create their first authoritative pair.');
}

var record = { current: current, maximum: maximum, sourceRef: sourceRef };
var effectType = mode === 'record' ? 'component.add' : 'component.set';
ctx.log(mode + ' Hit Points ' + current + '/' + maximum + '.');

return {
  narration: subject.name + "'s Hit Points are " + (mode === 'record' ? 'recorded' : 'corrected') + ' as ' + current + '/' + maximum + '.',
  effects: [{
    type: effectType,
    entityId: subject.id,
    definitionId: 'dnd2024.hit-points',
    data: JSON.stringify(record)
  }],
  data: {
    mode: mode,
    current: current,
    maximum: maximum,
    previous: previousPair,
    sourceRef: sourceRef
  }
};
